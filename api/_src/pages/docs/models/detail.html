模型
在 API 中使用

把请求中的 model 设为 __MODEL_ID__，Base URL 设为 https://cn.nexusvai.xyz/v1，就能用 OpenAI SDK 调用。同一个模型也可以通过 Responses 和 Messages（Anthropic 协议）接口调用。

import os
from openai import OpenAI

client = OpenAI(base_url="https://cn.nexusvai.xyz/v1", api_key=os.environ.get("CANCRI_API_KEY"))

response = client.chat.completions.create(
    model="__MODEL_ID__",
    messages=[{"role": "user", "content": "用一句话介绍你自己"}],
)
print(response.choices[0].message.content)
import OpenAI from "openai";

const client = new OpenAI({ baseURL: "https://cn.nexusvai.xyz/v1", apiKey: process.env.CANCRI_API_KEY });

const response = await client.chat.completions.create({
  model: "__MODEL_ID__",
  messages: [{ role: "user", content: "用一句话介绍你自己" }],
});
console.log(response.choices[0].message.content);
curl https://cn.nexusvai.xyz/v1/chat/completions \
  -H "Content-Type: application/json" \
  -H "Authorization: Bearer $CANCRI_API_KEY" \
  -d '{
    "model": "__MODEL_ID__",
    "messages": [{"role": "user", "content": "用一句话介绍你自己"}]
  }'
